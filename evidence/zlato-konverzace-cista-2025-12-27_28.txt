import { NextResponse } from "next/server";

export const revalidate = 60; // cache na serveru (1× za minutu)

async function fetchGoldUsdPerOz(): Promise<number> {
  const tryUrls = [
    "https://api.metals.live/v1/spot/gold",
    "https://api.metals.live/v1/spot",
  ];

  for (const url of tryUrls) {
    const r = await fetch(url, { cache: "no-store" });
    if (!r.ok) continue;
    const j: any = await r.json();

    if (Array.isArray(j)) {
      const first = j[0];

      // [["gold", price, ts]]
      if (Array.isArray(first)) {
        const p = Number(first?.[1]);
        if (Number.isFinite(p) && p > 0) return p;
      }

      // /spot: [["gold", ...], ["silver", ...]]
      const goldRow = j.find((x: any) => Array.isArray(x) && String(x?.[0]).toLowerCase() === "gold");
      if (goldRow) {
        const p = Number(goldRow?.[1]);
        if (Number.isFinite(p) && p > 0) return p;
      }

      // [{gold: price}]
      const obj0 = j[0];
      if (obj0 && typeof obj0 === "object" && !Array.isArray(obj0)) {
        const p = Number(obj0?.gold ?? obj0?.XAU ?? obj0?.price);
        if (Number.isFinite(p) && p > 0) return p;
      }
    }
  }

  throw new Error("Nepodařilo se načíst spot cenu zlata (USD/oz).");
}

async function fetchUsdCzk(): Promise<number> {
  const url =
    "https://cdn.jsdelivr.net/npm/@fawazahmed0/currency-api@latest/v1/currencies/usd.json";
  const r = await fetch(url, { cache: "no-store" });
  if (!r.ok) throw new Error("Nepodařilo se načíst kurz USD/CZK.");
  const j: any = await r.json();
  const rate = Number(j?.usd?.czk);
  if (!Number.isFinite(rate) || rate <= 0) throw new Error("Kurz USD/CZK je neplatný.");
  return rate;
}

export async function GET() {
  try {
    const [usdPerOz, usdCzk] = await Promise.all([fetchGoldUsdPerOz(), fetchUsdCzk()]);
    const czkPerOz = usdPerOz * usdCzk;

    return NextResponse.json({
      ok: true,
      usdPerOz,
      usdCzk,
      czkPerOz,
      ts: Date.now(),
    });
  } catch (e: any) {
    return NextResponse.json(
      { ok: false, error: String(e?.message || "Failed to load gold data") },
      { status: 500 }
    );
  }
}

}
